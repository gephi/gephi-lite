Use the browser console to test the driver. Example:


const driver = new GephiLiteDriver();
await driver.openGephiLite({baseUrl:'/gephi-lite/'});
await driver.importGraph({
  attributes: {},
  options: {},
  nodes: [
    { key: "A", attributes: { label:"A" } },
    { key: "B", attributes: { label:"B" } },
    { key: "C", attributes: { label:"C" } }
  ],
  edges: [
    { key:"AB", source:"A", target:"B"},
    { key:"BC", source:"B", target:"C"},
    { key:"CA", source:"C", target:"A"},
  ]
});
await driver.setSelection({nodeIds:[], edgeIds:["AB"]});
